# =============================================================================
# FILE    : src/backend/requirements.txt
# PROJECT : IDRM (Integrated Disaster Response Management) — MVP
#
# PURPOSE : Lists all Python packages installed via pip inside the
#           "idrm-mvp" conda environment.
#
# ── HOW TO USE ───────────────────────────────────────────────────────────────
#
#   PREFERRED — let conda handle everything (reads this file automatically):
#     conda env create  -f environment.yml          # first time
#     conda env update  -f environment.yml --prune  # after pulling new code
#
#   MANUAL — re-install pip packages inside an already-active conda env:
#     conda activate idrm-mvp
#     pip install -r src/backend/requirements.txt
#
#   CHECK what is installed:
#     conda activate idrm-mvp && pip list
#
# ── WHERE DO THE GEOSPATIAL LIBRARIES LIVE? ──────────────────────────────────
#
#   SHAPELY  →  NOT here — it is in environment.yml (conda tier).
#               Reason: shapely bundles a GEOS C library. Installing it via
#               conda-forge guarantees binary compatibility with the rest of
#               the conda environment. Installing it here via pip could cause
#               GEOS version conflicts on machines that also have QGIS or
#               other geospatial tools installed.
#
#   GEOALCHEMY2  →  HERE (pip tier).
#               Reason: GeoAlchemy2 is pure Python — it contains no C code.
#               It simply teaches SQLAlchemy to understand PostGIS column
#               types (GEOMETRY, GEOGRAPHY) and functions (ST_Distance, etc.).
#               Pure Python packages are always fine to install via pip.
#
# ── VERSION POLICY ────────────────────────────────────────────────────────────
#
#   We use >= (minimum bound) so pip installs the latest compatible version.
#   This means development environments always get security updates.
#
#   For a fully reproducible production deployment, generate a lock file:
#
#     conda activate idrm-mvp
#     pip freeze > requirements.lock.txt
#
#   Then deploy using the lock file:
#     pip install -r requirements.lock.txt
#
#   ⚠ NEVER commit .lock.txt to git unless it is the production lock.
#     Development lock files contain absolute paths and are not portable.
#
# ── ADDING A NEW PACKAGE ─────────────────────────────────────────────────────
#
#   1. Decide: does it have a C native dependency? → conda. Otherwise → pip.
#   2. Add it to this file with a comment explaining what it does.
#   3. Also add it to environment.yml under the pip: section.
#   4. Run: pip install -r src/backend/requirements.txt  (inside idrm-mvp env)
#   5. Test: conda activate idrm-mvp && python -c "import <package>"
#
# =============================================================================

# FastAPI: async web framework for building REST APIs with Python type hints.
# Auto-generates OpenAPI (Swagger) docs at http://localhost:8000/docs
# Latest stable: 0.136.x (April 2026). We require >= 0.115.0 (LTS-era).
fastapi>=0.115.0

# uvicorn: ASGI server that runs FastAPI.
# [standard] extras add:
#   watchfiles   — auto-reloads the server when you save a .py file
#   websockets   — WebSocket support (used by IDRM live alerts)
#   httptools    — faster HTTP parser (C extension, has PyPI binary wheel)
uvicorn[standard]>=0.30.0


# ─────────────────────────────────────────────────────────────────────────────
# DATABASE — ORM & DRIVERS
# ─────────────────────────────────────────────────────────────────────────────

# SQLAlchemy 2.x: Python ORM (Object-Relational Mapper).
# Maps Python classes to PostgreSQL tables.
# [asyncio] extra: enables async/await syntax for all database operations.
# ⚠ SQLAlchemy 2.x API is NOT backward compatible with 1.x.
sqlalchemy[asyncio]>=2.0.30

# asyncpg: async PostgreSQL driver.
# SQLAlchemy uses asyncpg under the hood when you use async sessions.
# Significantly faster than psycopg2 for concurrent requests.
asyncpg>=0.29.0

# psycopg2-binary: synchronous PostgreSQL driver.
# Alembic (migrations) cannot use async drivers, so it needs psycopg2.
# The -binary variant bundles libpq so no system postgresql-dev headers needed.
# ⚠ Production best practice: use psycopg2 (without -binary) and install
#   postgresql-server-dev-16 from apt for a proper compiled version.
psycopg2-binary>=2.9.9


# ─────────────────────────────────────────────────────────────────────────────
# DATABASE — GEOSPATIAL EXTENSION (PostGIS)
# ─────────────────────────────────────────────────────────────────────────────

# GeoAlchemy2: PostGIS support for SQLAlchemy.
# Adds spatial column types: Geometry, Geography, Raster.
# Adds PostGIS functions you can use in SQLAlchemy queries:
#   ST_Distance, ST_Within, ST_DWithin, ST_AsGeoJSON, etc.
#
# WHY pip (not conda)?
#   GeoAlchemy2 is 100% pure Python — no C/C++ code, no native library deps.
#   It delegates all geometry work to PostGIS (in the database) and to
#   shapely (already in conda). Pure Python = pip is the right installer.
geoalchemy2>=0.14.2

# NOTE: shapely is NOT listed here because it is managed by conda-forge.
# See environment.yml for shapely>=2.0.6 under the conda dependencies section.
# Running `pip install shapely` would risk installing a GEOS version that
# conflicts with the one conda-forge already set up.


# ─────────────────────────────────────────────────────────────────────────────
# DATABASE — MIGRATIONS
# ─────────────────────────────────────────────────────────────────────────────

# Alembic: database schema migration tool for SQLAlchemy.
# Every change to a SQLAlchemy model must have a matching Alembic migration
# so the change can be applied to the database in a controlled, reversible way.
#
# Common commands (run inside the idrm-mvp env):
#   alembic init alembic                        — one-time init
#   alembic revision --autogenerate -m "msg"    — create new migration
#   alembic upgrade head                        — apply all pending migrations
#   alembic downgrade -1                        — roll back one migration
alembic>=1.13.0
